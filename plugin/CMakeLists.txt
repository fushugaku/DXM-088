cmake_minimum_required(VERSION 3.22)
project(DXM-088 VERSION 0.9.0 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(IPLUG2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/iplug2" CACHE PATH "iPlug2 SDK")
set(IPLUG_DEPLOY_PLUGINS OFF CACHE BOOL "Install with tools/install-plugins.sh" FORCE)
include(${IPLUG2_DIR}/iPlug2.cmake)
find_package(iPlug2 REQUIRED)
set(LXR_RUST_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/engine/target/release/liblxr02_plugin.a" CACHE FILEPATH "Rust DSP library")
add_library(lxr-rust STATIC IMPORTED)
set_target_properties(lxr-rust PROPERTIES IMPORTED_LOCATION "${LXR_RUST_LIBRARY}")
file(GLOB_RECURSE LXR_WEB CONFIGURE_DEPENDS "resources/web/*")
set(LXR_FORMATS VST3 CLAP)
if(APPLE)
  list(APPEND LXR_FORMATS AU)
endif()
if(APPLE)
  set(LXR_DIALOG file_dialog_mac.mm)
  set_source_files_properties(file_dialog_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
elseif(WIN32)
  set(LXR_DIALOG file_dialog_win.cpp)
endif()
iplug_add_plugin(DXM-088 SOURCES LXR02.cpp LXR02.h KitStore.h engine.h ${LXR_DIALOG} WEB_RESOURCES ${LXR_WEB} UI WEBVIEW FORMATS ${LXR_FORMATS} LINK lxr-rust)
# SDK translation units do not include the product config.h. Set these for the
# entire target so the processor bus and controller mapping agree on 16 channels.
target_compile_definitions(DXM-088-vst3 PRIVATE VST3_NUM_MIDI_IN_CHANS=16 VST3_NUM_CC_CHANS=16)
if(APPLE)
  foreach(format vst3 clap au)
    # iPlug derives Objective-C symbols from the product name. Hyphens belong
    # in the displayed name and filenames, but not in Objective-C identifiers.
    get_target_property(objc_definitions DXM-088-${format} COMPILE_DEFINITIONS)
    list(FILTER objc_definitions EXCLUDE REGEX "^OBJC_PREFIX=")
    list(APPEND objc_definitions OBJC_PREFIX=vDXM088)
    set_property(TARGET DXM-088-${format} PROPERTY COMPILE_DEFINITIONS ${objc_definitions})
    target_link_libraries(DXM-088-${format} PRIVATE "-framework Security" "-framework SystemConfiguration" "-framework CoreFoundation" "-liconv")
  endforeach()
endif()
if(WIN32)
  add_executable(dxm088-kit-store-tests ../tools/kit-store-tests.cpp)
  target_compile_features(dxm088-kit-store-tests PRIVATE cxx_std_17)
  target_compile_definitions(dxm088-kit-store-tests PRIVATE OS_WIN=1)
  target_compile_options(dxm088-kit-store-tests PRIVATE /utf-8)
  target_include_directories(dxm088-kit-store-tests PRIVATE "${IPLUG2_DIR}/Dependencies/Extras/nlohmann")
  foreach(format vst3 clap)
    target_link_libraries(DXM-088-${format} PRIVATE ws2_32 userenv bcrypt ntdll comdlg32)
    # Windows resources are files beside the module, not macOS bundle resources.
    if(format STREQUAL "vst3")
      set(editor_dir "${CMAKE_BINARY_DIR}/out/DXM-088.vst3/Contents/Resources/web")
    else()
      set(editor_dir "${CMAKE_BINARY_DIR}/out/DXM-088.resources/web")
    endif()
    add_custom_command(TARGET DXM-088-${format} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/resources/web" "${editor_dir}")
  endforeach()
endif()
